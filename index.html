// ============ АНАЛИТИКА ============ function renderAnalytics() { var deals = getAnalyticsDeals(); var totalDeals = deals.length; var failedDealsArr = deals.filter(function (d) { return d.stage === 'failed'; }); var failedDeals = failedDealsArr.length; var failedAmount = failedDealsArr.reduce(function (sum, d) { return sum + (d.investmentAmount || 0); }, 0); var totalInvestmentRaw = deals.reduce(function (sum, d) { return sum + (d.investmentAmount || 0); }, 0); var totalInvestment = totalInvestmentRaw - failedAmount; var fundedDealsArr = deals.filter(function (d) { return d.stage === 'funded'; }); var fundedDeals = fundedDealsArr.length; var fundedAmount = fundedDealsArr.reduce(function (sum, d) { return sum + (d.investmentAmount || 0); }, 0); var auditReachedArr = deals.filter(function (d) { return d.stage === 'audit' || d.stage === 'funded'; }); var auditReached = auditReachedArr.length; var conversionAudit = totalDeals > 0 ? ((auditReached / totalDeals) * 100).toFixed(1) : '0.0'; var conversionFunded = totalDeals > 0 ? ((fundedDeals / totalDeals) * 100).toFixed(1) : '0.0'; var scopeLabel = state.analyticsScope === 'mine' ? 'Моя статистика' : 'Общая статистика команды'; return `

${scopeLabel}

Показано сделок: ${totalDeals} из ${state.deals.length}

Всего сделок

${totalDeals}

Факапов: ${failedDeals}

Общий объём без факапов

${formatMoney(totalInvestment)}

Исключено факапов: ${formatMoney(failedAmount)}

Получено

${formatMoney(fundedAmount)}

Успешных сделок: ${fundedDeals}

Конверсия в аудит

${conversionAudit}%

Дошли до аудита: ${auditReached} из ${totalDeals}

Конверсия в инвестиции

${conversionFunded}%

Инвестиции получены: ${fundedDeals} из ${totalDeals}

Провалено

${failedDeals}

Сумма факапов: ${formatMoney(failedAmount)}

Распределение по этапам

Объём инвестиций (млн €)

Эффективность менеджеров

`; }
